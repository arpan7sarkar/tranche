# Foundation (F0 + F1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Prove which PayPal sandbox capabilities we can use, then freeze the shared contracts and database schema that both lanes build against.

**Architecture:** `packages/contracts` holds every shape that crosses a lane boundary as Zod schemas plus a few pure helpers. `packages/db` holds the Drizzle schema, split into `schema/core.ts` (owned by Person H) and `schema/agents.ts` (owned by Person E). The only write path from agents into money is the `proposed_action` table; the only read path from money into agents is the `event` table.

**Tech Stack:** TypeScript 7, Zod 4.6, Drizzle ORM 0.45 + drizzle-kit 0.31, postgres.js 3.4, Vitest 5, Postgres 18, tsx.

**Spec:** `docs/superpowers/specs/2026-10-08-tranche-design.md`

**Owner:** Person H. Person E reviews the contracts PR (Task 2) before it merges, because the E lane depends on it.

## Global Constraints

- PayPal sandbox only; base URL `https://api-m.sandbox.paypal.com`.
- Money is stored as integer minor units plus an ISO 4217 code; PayPal string amounts exist only at the `packages/paypal` boundary.
- HUF, JPY and TWD have no decimals at PayPal; every other supported currency has 2.
- Dates are ISO strings in contracts (`YYYY-MM-DD` for calendar dates, ISO 8601 with offset for instants).
- No em dash characters anywhere; use a plain `-`.
- Comments only when they carry real value.
- Conventional commits, no AI co-author lines.

## Review Focus

- **Plan with a dependency cycle** (task A depends on B, B on A): `planIssues` must report it, and `PlanDraft.parse` must reject it - covered in Task 2 Step 1.
- **Payee shares that do not sum to 100%** (for example 60% + 30%): rejected - covered in Task 2 Step 1.
- **PayPal amount strings with 3 decimals or a zero-decimal currency with cents** (`"10.005"`, JPY `"10.50"`): `fromPayPalValue` throws instead of silently rounding - covered in Task 2 Step 1.
- **The same webhook or the same agent proposal delivered twice**: the unique indexes on `inbox(source, source_event_id)` and `proposed_action(idempotency_key)` reject the duplicate - covered in Task 3 Step 1.
- **A milestone with no machine checks**: `AcceptanceCriteria` requires at least one check, so a release can never be justified by prose alone - covered in Task 2 Step 1.

const BASE = "https://api-m.sandbox.paypal.com";

function need(name: string): string {
  const v = process.env[name];
  if (!v) throw new Error(`${name} is not set`);
  return v;
}

const clientId = need("PAYPAL_CLIENT_ID");
const secret = need("PAYPAL_CLIENT_SECRET");
const agency = need("PROBE_AGENCY_EMAIL");
const sub = need("PROBE_SUB_EMAIL");

type Result = { capability: string; ok: boolean; status: number; detail: unknown };
const results: Result[] = [];

async function token(): Promise<{ accessToken: string; scopes: string[] }> {
  const res = await fetch(`${BASE}/v1/oauth2/token`, {
    method: "POST",
    headers: {
      Authorization: `Basic ${Buffer.from(`${clientId}:${secret}`).toString("base64")}`,
      "Content-Type": "application/x-www-form-urlencoded",
    },
    body: "grant_type=client_credentials",
  });
  const body = (await res.json()) as { access_token: string; scope: string };
  if (!res.ok) throw new Error(`token failed: ${res.status} ${JSON.stringify(body)}`);
  return { accessToken: body.access_token, scopes: body.scope.split(" ") };
}

async function call(capability: string, accessToken: string, path: string, body: unknown) {
  const res = await fetch(`${BASE}${path}`, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${accessToken}`,
      "Content-Type": "application/json",
      "PayPal-Request-Id": crypto.randomUUID(),
    },
    body: JSON.stringify(body),
  });
  const detail = await res.json().catch(() => null);
  results.push({ capability, ok: res.ok, status: res.status, detail });
}

const amount = (value: string) => ({ currency_code: "USD", value });

const { accessToken, scopes } = await token();
console.log("SCOPES");
for (const s of scopes) console.log(`  ${s}`);

await call("partner_referrals", accessToken, "/v2/customer/partner-referrals", {
  tracking_id: `probe-${Date.now()}`,
  operations: [

      operation: "API_INTEGRATION",
      api_integration_preference: {
        rest_api_integration: {
          integration_method: "PAYPAL",
          integration_type: "THIRD_PARTY",
          third_party_details: {
            features: ["PAYMENT", "REFUND", "PARTNER_FEE", "DELAY_FUNDS_DISBURSEMENT", "VAULT"],

      },

  products: ["EXPRESS_CHECKOUT"],
  legal_consents: [{ type: "SHARE_DATA_CONSENT", granted: true }],
});

await call("order_authorize_baseline", accessToken, "/v2/checkout/orders", {
  intent: "AUTHORIZE",

});

await call("order_delayed_disbursement_with_fee", accessToken, "/v2/checkout/orders", {
  intent: "CAPTURE",
  purchase_units: [
    {
      amount: amount("100.00"),
      payee: { email_address: agency },
      payment_instruction: {
        disbursement_mode: "DELAYED",
        platform_fees: [{ amount: amount("5.00") }],
      },
    },
  ],
});

await call("order_multiple_payees", accessToken, "/v2/checkout/orders", {
  intent: "CAPTURE",
  purchase_units: [
    { reference_id: "agency", amount: amount("80.00"), payee: { email_address: agency } },

`packages/contracts/src/money.ts`:

```ts
import { z } from "zod";

export const CurrencyCode = z.string().regex(/^[A-Z]{3}$/);
export const Money = z.object({
  currency: CurrencyCode,
  minor: z.number().int().nonnegative(),
});
export type Money = z.infer<typeof Money>;

const ZERO_DECIMAL = new Set(["HUF", "JPY", "TWD"]);

export function minorUnitExponent(currency: string): 0 | 2 {
  return ZERO_DECIMAL.has(currency) ? 0 : 2;
}

export function toPayPalValue(m: Money): string {
  if (minorUnitExponent(m.currency) === 0) return String(m.minor);
  const s = String(m.minor).padStart(3, "0");
  return `${s.slice(0, -2)}.${s.slice(-2)}`;
}

export function fromPayPalValue(currency: string, value: string): Money {
  const exponent = minorUnitExponent(currency);
  const pattern = exponent === 0 ? /^\d+$/ : /^\d+(\.\d{1,2})?$/;
  if (!pattern.test(value)) throw new Error(`Cannot represent ${value} ${currency} exactly`);
  if (exponent === 0) return { currency, minor: Number(value) };
  const [whole = "0", fraction = ""] = value.split(".");
  return { currency, minor: Number(whole) * 100 + Number(fraction.padEnd(2, "0")) };
}

export function addMoney(a: Money, b: Money): Money {
  if (a.currency !== b.currency) throw new Error(`Currency mismatch: ${a.currency} vs ${b.currency}`);
  return { currency: a.currency, minor: a.minor + b.minor };
}

  currency: CurrencyCode,
  milestones: z.array(PlanMilestone).min(1).max(12),
  tasks: z.array(PlanTask).min(1).max(80),
});

export function planIssues(plan: z.infer<typeof PlanShape>): string[] {
  const issues: string[] = [];
  const taskKeys = new Set<string>();
  const milestoneKeys = new Set<string>();

  for (const m of plan.milestones) {
    if (milestoneKeys.has(m.key)) issues.push(`duplicate milestone key "${m.key}"`);
    milestoneKeys.add(m.key);
    if (m.price.currency !== plan.currency) {
      issues.push(`milestone "${m.key}" is priced in ${m.price.currency}, plan currency is ${plan.currency}`);
    }
    const bps = m.payees.reduce((sum, p) => sum + p.shareBps, 0);
    if (bps !== 10000) issues.push(`milestone "${m.key}" payee shares sum to ${bps} bps, expected 10000`);
  }

  for (const t of plan.tasks) {
    if (taskKeys.has(t.key)) issues.push(`duplicate task key "${t.key}"`);
    taskKeys.add(t.key);
  }

  for (const t of plan.tasks) {
    if (!milestoneKeys.has(t.milestoneKey)) {
      issues.push(`task "${t.key}" references unknown milestone "${t.milestoneKey}"`);
    }
    for (const d of t.dependsOn) {
      if (!taskKeys.has(d)) issues.push(`task "${t.key}" depends on unknown task "${d}"`);
    }
  }

  for (const m of plan.milestones) {
    if (!plan.tasks.some((t) => t.milestoneKey === m.key)) issues.push(`milestone "${m.key}" has no tasks`);
  }

  const indegree = new Map(plan.tasks.map((t) => [t.key, 0]));
  for (const t of plan.tasks) {
    for (const d of t.dependsOn) if (taskKeys.has(d)) indegree.set(t.key, (indegree.get(t.key) ?? 0) + 1);
  }
  const queue = [...indegree].filter(([, n]) => n === 0).map(([k]) => k);
  let visited = 0;
  while (queue.length > 0) {
    const k = queue.shift() as string;
    visited++;
    for (const t of plan.tasks) {
      if (!t.dependsOn.includes(k)) continue;
      const n = (indegree.get(t.key) ?? 0) - 1;
      indegree.set(t.key, n);
      if (n === 0) queue.push(t.key);
    }
  }
  if (visited < indegree.size) issues.push("tasks contain a dependency cycle");

  return issues;
}

export const PlanDraft = PlanShape.superRefine((plan, ctx) => {
  for (const message of planIssues(plan)) ctx.addIssue({ code: "custom", message });
});
export type PlanDraft = z.infer<typeof PlanDraft>;
```

import { createHash, randomBytes, timingSafeEqual } from "node:crypto";
import { eq } from "drizzle-orm";
import type { Db } from "./index";
import { magicLink } from "./schema";

export type Tx = Parameters<Parameters<Db["transaction"]>[0]>[0];

const TOKEN = /^[A-Za-z0-9_-]{43}$/;

export function hashToken(token: string): string {
  return createHash("sha256").update(token).digest("hex");
}

export async function createMagicLink(
  db: Db | Tx,
  input: { clientId: string; dealId: string; ttlDays?: number; now?: Date },
): Promise<{ token: string; url: string }> {
  const token = randomBytes(32).toString("base64url");
  const now = input.now ?? new Date();
  await db.insert(magicLink).values({
    clientId: input.clientId,
    dealId: input.dealId,
    tokenHash: hashToken(token),
    expiresAt: new Date(now.getTime() + (input.ttlDays ?? 30) * 86_400_000),
  });
  return { token, url: `${process.env.APP_URL ?? "http://localhost:3000"}/portal/${token}` };
}

export async function resolveMagicLink(db: Db, token: string, now = new Date()) {
  if (!TOKEN.test(token)) return null;
  const hash = hashToken(token);
  const [row] = await db.select().from(magicLink).where(eq(magicLink.tokenHash, hash));
  if (!row) return null;
  if (!timingSafeEqual(Buffer.from(row.tokenHash, "hex"), Buffer.from(hash, "hex"))) return null;
  if (row.expiresAt <= now) return null;
  return { clientId: row.clientId, dealId: row.dealId };
}
